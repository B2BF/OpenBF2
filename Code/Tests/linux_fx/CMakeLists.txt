# [TEMP] [CHOSEN, not recovered] Linux bytecode and CPU effect contracts; issue #36.
cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(BF2LinuxEffects LANGUAGES CXX)
    enable_testing()
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Linux effect checks require a 64-bit Linux toolchain")
endif()
get_filename_component(BF2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../BF2" ABSOLUTE)
set(BF2_LINUX_EFFECT_CHECKS ON)
if(NOT TARGET bf2_fx_native_compiler)
    add_subdirectory("${BF2_ROOT}/Platform/Gfx/Effect" "${CMAKE_CURRENT_BINARY_DIR}/effect")
endif()
add_executable(bf2_linux_fx_runtime ../macos_fx_runtime/main.cpp)
target_link_libraries(bf2_linux_fx_runtime PRIVATE bf2_fx_runtime)
add_executable(bf2_linux_fx_compiler compiler.cpp)
target_link_libraries(bf2_linux_fx_compiler PRIVATE bf2_fx_native_compiler)
add_executable(bf2_linux_fx_equivalence equivalence.cpp)
target_link_libraries(bf2_linux_fx_equivalence PRIVATE bf2_fx_native_compiler)
add_dependencies(bf2_linux_fx_equivalence bf2_fx_equivalence)
# State-only effects need IUnknown lifetime, with every graphics method failing visibly.
# Generate those unused overrides from the public subset instead of duplicating its signatures.
file(READ "${BF2_ROOT}/Platform/Gfx/D3D9Subset.h" _state_device_header)
string(REGEX MATCH "struct IDirect3DDevice9 : IUnknown \\{([^}]*)\\};" _state_device_match "${_state_device_header}")
if(NOT _state_device_match)
    message(FATAL_ERROR "Cannot find the state observer's public device interface")
endif()
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/state-device-methods.inc" CONTENT "${CMAKE_MATCH_1}" @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${BF2_ROOT}/Platform/Gfx/D3D9Subset.h")
add_executable(bf2_linux_fx_packed_colors packed_colors.cpp)
target_include_directories(bf2_linux_fx_packed_colors PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(bf2_linux_fx_packed_colors PRIVATE bf2_dxvk_effect)
foreach(_target bf2_linux_fx_runtime bf2_linux_fx_compiler bf2_linux_fx_equivalence bf2_linux_fx_packed_colors)
    set_target_properties(${_target} PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
    target_compile_definitions(${_target} PRIVATE DEDICATED=0 BF2_IMAGE_RENDDX9=1 BF2_GFX_DXVK=1)
    target_compile_options(${_target} PRIVATE "SHELL:-include ${BF2_ROOT}/Core/BuildConfig.h"
        -Wall -Wextra -Werror $<$<CXX_COMPILER_ID:GNU>:-Wno-misleading-indentation>)
endforeach()
add_test(NAME linux_fx_runtime COMMAND bf2_linux_fx_runtime "${CMAKE_CURRENT_BINARY_DIR}/scratch")
add_test(NAME linux_fx_compiler COMMAND bf2_linux_fx_compiler "${CMAKE_CURRENT_BINARY_DIR}/scratch")
add_test(NAME linux_fx_equivalence COMMAND bf2_linux_fx_equivalence "${CMAKE_CURRENT_BINARY_DIR}/equivalence-scratch")
add_test(NAME linux_fx_packed_colors COMMAND bf2_linux_fx_packed_colors "${CMAKE_CURRENT_BINARY_DIR}/packed-colors-scratch")
set_tests_properties(linux_fx_equivalence PROPERTIES FIXTURES_SETUP native_equivalence_inputs)
foreach(_case positive initializer invalid)
    add_test(NAME linux_fx_equivalence_${_case} COMMAND bf2_fx_equivalence
        "${CMAKE_CURRENT_BINARY_DIR}/equivalence-scratch/${_case}.list"
        "${CMAKE_CURRENT_BINARY_DIR}/equivalence-scratch/${_case}.manifest" 1)
    set_tests_properties(linux_fx_equivalence_${_case} PROPERTIES FIXTURES_REQUIRED native_equivalence_inputs)
    if(NOT _case STREQUAL "positive")
        set_tests_properties(linux_fx_equivalence_${_case} PROPERTIES WILL_FAIL TRUE)
    endif()
endforeach()

option(BF2_LINUX_FX_DEVICE_CHECK "Build the manually run SDL/DXVK effect rendering check" OFF)
if(BF2_LINUX_FX_DEVICE_CHECK)
    find_package(SDL3 3.2 REQUIRED CONFIG)
    if(NOT TARGET bf2_dxvk_forward)
        add_subdirectory(../linux_vulkan "${CMAKE_CURRENT_BINARY_DIR}/vulkan")
    endif()
    add_executable(bf2_linux_fx_device device.cpp)
    target_compile_definitions(bf2_linux_fx_device PRIVATE DEDICATED=0 BF2_GFX_DXVK=1)
    target_compile_options(bf2_linux_fx_device PRIVATE "SHELL:-include ${BF2_ROOT}/Core/BuildConfig.h" -Wall -Wextra -Werror)
    target_link_libraries(bf2_linux_fx_device PRIVATE bf2_dxvk_effect bf2_dxvk_forward SDL3::SDL3)
    set_target_properties(bf2_linux_fx_device PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
    get_filename_component(_dxvk_libdir "${BF2_DXVK_LIBRARY}" DIRECTORY)
    set_target_properties(bf2_linux_fx_device PROPERTIES BUILD_RPATH "${_dxvk_libdir}")
endif()

function(bf2_arm_test target dedicated)
    target_compile_definitions(${target} PRIVATE DEDICATED=${dedicated} BF2_LINUX_CLIENT=1
        BF2_PLATFORM_POSIX=1 BF2_NATIVE_X87=1 BF2_STATIC_PLUGINS=1 _ITERATOR_DEBUG_LEVEL=0)
    target_compile_options(${target} PRIVATE -ffp-contract=off -frounding-math -fno-strict-aliasing
        "-include${BF2_ARM_TEST_ROOT}/BF2/Core/BuildConfig.h"
        "-include${BF2_ARM_TEST_ROOT}/BF2/Core/PlatformCompat.h")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
        # Match the native client's explicit-self vtable adapter compile contract.
        target_compile_options(${target} PRIVATE -fno-devirtualize -fno-devirtualize-speculatively)
    endif()
    set_target_properties(${target} PROPERTIES CXX_STANDARD 14 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
    add_test(NAME linux_${target} COMMAND ${target})
endfunction()
